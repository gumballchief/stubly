    // The six calls that move a job through its life.
async function createJob(client, { providerAddr, evaluatorAddr, expiresInSec, description })
async function setBudget(providerSigner, jobId, amount)
async function fund(clientSigner, jobId, amount)
async function submit(providerSigner, jobId, deliverableText)
async function complete(evaluatorSigner, jobId, reasonText)   // pays the agent
async function reject(evaluatorSigner, jobId, reasonText)     // refunds the client

// Every write is simulated before it is sent, so a mistake costs a log line, not gas.
await contract[method].staticCall(...args);
    Say this “These are the six calls that move a job from created to settled. Create the job, the agent quotes a price, the client funds the escrow, the agent submits the work, and then the evaluator either completes it — which pays the agent — or rejects it, which refunds the buyer. That last line simulates every transaction before sending it, so if something would fail, it costs me a log line instead of gas.”
